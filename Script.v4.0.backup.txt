    // directOnly=true checks only el's direct text-node children
    // (e.g. preview text sitting right before a nested tooltip),
    // rather than walking the full subtree — avoids redundantly
    // re-processing a nested .recipe_tooltip that's already handled
    // by its own call.
    function textNodesOf(el, directOnly) {
        if (directOnly) {
            return Array.from(el.childNodes).filter((n) => n.nodeType === 3);
        }
        const out = [];
        const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT);
        let n;
        while ((n = walker.nextNode())) out.push(n);
        return out;
    }

    function applyItemNames(selector, directOnly) {
        if (!ENABLE_ITEM_NAMES) return;
        for (const el of document.querySelectorAll(selector)) {
            for (const node of textNodesOf(el, directOnly)) {
                const original = node.textContent;
                const trimmed = original.trim();

                // Whole node is exactly an item name.
                if (itemNames[trimmed]) {
                    node.textContent = original.replace(trimmed, itemNames[trimmed]);
                    el.classList.add('translated');
                    continue;
                }
                // "ItemName, 140%, x1" (selection previews) — exact
                // text before the first comma.
                const commaIdx = trimmed.indexOf(',');
                if (commaIdx > -1) {
                    const candidate = trimmed.slice(0, commaIdx).trim();
                    if (itemNames[candidate]) {
                        node.textContent = original.replace(candidate, itemNames[candidate]);
                        el.classList.add('translated');
                        continue;
                    }
                }
                // "ItemName x0/4" (materials lists) — exact text before
                // a space-then-"x"-then-digit boundary.
                const spaceX = trimmed.match(/^(.+?)\s+x\d/);
                if (spaceX && itemNames[spaceX[1]]) {
                    node.textContent = original.replace(spaceX[1], itemNames[spaceX[1]]);
                    el.classList.add('translated');
                    continue;
                }
                // "[ItemName]+[ItemName]+..." (assembled-equipment
                // component list) — any number of bracketed segments;
                // each one is checked independently, so a bracket that
                // isn't a recognized item name is left as-is rather
                // than skipping the whole line.
                if (/\[[^[\]]+\]/.test(trimmed)) {
                    const swapped = original.replace(/\[([^[\]]+)\]/g, (whole, inner) => {
                        const t = itemNames[inner.trim()];
                        return t ? `[${t}]` : whole;
                    });
                    if (swapped !== original) {
                        node.textContent = swapped;
                        el.classList.add('translated');
                    }
                }
                // '"ItemName" x1, "ItemName" x1' (message/event log
                // entries) — same idea as the bracket pattern above,
                // just quote-delimited instead.
                if (/"[^"]+"/.test(trimmed)) {
                    const swapped2 = node.textContent.replace(/"([^"]+)"/g, (whole, inner) => {
                        const t = itemNames[inner.trim()];
                        return t ? `"${t}"` : whole;
                    });
                    if (swapped2 !== node.textContent) {
                        node.textContent = swapped2;
                        el.classList.add('translated');
                    }
                }
                // "使用 ItemName , ..." (single gem-use messages) — the
                // item name sits between spaces before a comma,
                // unlike the other patterns where the name starts at
                // the very beginning of the text. \s* (not \s) since
                // a same-pass buttons-mechanism cleanup pair may have
                // already tightened " , " to ", " by the time this
                // runs. Matches any non-whitespace run before the
                // comma, not just after a known prefix — safe
                // regardless, since it still has to be an exact
                // dictionary hit to do anything.
                const spaceComma = node.textContent.match(/(\S+)\s*,/);
                if (spaceComma && itemNames[spaceComma[1]]) {
                    node.textContent = node.textContent.replace(spaceComma[1], itemNames[spaceComma[1]]);
                    el.classList.add('translated');
                }
                // "..., ItemName." (bulk-use messages) — reverse of the
                // pattern above, item name follows the comma instead
                // of preceding it, ending at a period. Verified this
                // doesn't misfire on the single-gem message's longer
                // ", Stat increased by N[Soft Cap]." tail, since \S+
                // can't cross the space inside that phrase to reach a
                // period — same safety net either way (exact
                // dictionary hit required).
                const commaThenItem = node.textContent.match(/,\s*(\S+)\./);
                if (commaThenItem && itemNames[commaThenItem[1]]) {
                    node.textContent = node.textContent.replace(commaThenItem[1], itemNames[commaThenItem[1]]);
                    el.classList.add('translated');
                }
                // "SkillName : level X/Y" (skill list bar labels) —
                // skill names reuse itemNames rather than a separate
                // dictionary, since they're just as discrete/unique as
                // item names and an exact-match hit is exact-match
                // regardless of which conceptual bucket the string
                // came from.
                const skillLevel = node.textContent.match(/^(.+?)\s*:\s*level/);
                if (skillLevel && itemNames[skillLevel[1]]) {
                    node.textContent = node.textContent.replace(skillLevel[1], itemNames[skillLevel[1]]);
                    el.classList.add('translated');
                }
            }
        }
    }

    // Escapes regex special characters so a plain string can be used
    // as a literal pattern with the global flag.
    function escapeRegex(s) {
        return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    }

    function applySelector(selector, pairs) {
        for (const el of document.querySelectorAll(selector)) {
            const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT);
            let node;
            while ((node = walker.nextNode())) {
                for (const [zh, en] of pairs) {
                    if (node.textContent.includes(zh)) {
                        // Global regex, not a plain string replace —
                        // a term like 点 can appear several times in
                        // one sentence (once per stat in a bulk-gain
                        // message), and .replace(str, val) only ever
                        // replaces the first occurrence.
                        node.textContent = node.textContent.replace(new RegExp(escapeRegex(zh), 'g'), en);
                        el.classList.add('translated');
                    }
                }
            }
        }
    }

    function scan() {
        // Skill names run before the buttons/substring pass below —
        // raw skill name text needs to hit the exact-match pattern
        // ("SkillName : level X/Y") before anything in #skill_list's
        // own substring-matched pairs gets a chance to partially
        // consume it. This is what caused 秘法精通 → "Technique精通"
        // and 武器精通 → "Weapon精通": the bare 秘法/武器 category-label
        // pairs fired first and corrupted the longer skill names
        // before exact-match ever saw the original string. Those bare
        // pairs are gone now (replaced with combined "X 技能" pairs
        // that can't collide), but running this first is a second,
        // independent safety layer against the same class of bug
        // recurring if more #skill_list buttons pairs get added later.
        applyItemNames('#skill_list');
        for (const [selector, pairs] of Object.entries(buttons)) applySelector(selector, pairs);
        // Item names inside recipe tooltips (result name + materials
        // list), for every crafting/cooking/component "items"-type
        // category, parts included.
        applyItemNames('.recipe_tooltip');
        // Also try recipe_name headers directly — for single-recipe
        // items (no material choice) the header IS the item name, so
        // this catches it. Safe to apply everywhere unconditionally:
        // it's exact-match, so recipe names that aren't item names
        // (cooking recipes, component categories) simply find no
        // match and are left untouched.
        applyItemNames('.recipe_name');
        // Preview text on each material option in a "parts" recipe
        // (sits right before that option's own nested tooltip, which
        // the .recipe_tooltip call above already covers separately —
        // directOnly avoids redoing that nested work here).
        applyItemNames('.selectable_material', true);
        // Same idea for the equipment-assembly dropdowns (Sword,
        // Helmet, etc.) — selectable_component's own preview text
        // ("ItemName, 140%, x1") uses the comma pattern already
        // handled above, so no new matching logic needed here.
        applyItemNames('.selectable_component', true);
        // Inventory: item_tooltip is a separate class from crafting's
        // recipe_tooltip but the same shape (bolded name up top).
        // Scoped under #inventory_content_div in case these class
        // names get reused in a shop/trade window elsewhere.
        // item_tooltip is reused across inventory AND the gear tab
        // (and possibly elsewhere), so this is intentionally unscoped
        // rather than tied to #inventory_content_div.
        applyItemNames('.item_tooltip');
        // Inventory: the compact visible name, which sits in one of
        // two wrapper classes depending on equipped vs regular —
        // nesting depth varies (sometimes an extra rarity_* span in
        // between) but the full-subtree walk handles that either way.
        // Same two wrapper classes as before, now confirmed reused in
        // the trader panel too, so unscoped rather than tied to
        // #inventory_content_div.
        applyItemNames('.equipped_item_name, .inventory_item_name');
        // Gear tab: visible name on each equipped slot. Direct-child
        // combinator so this only catches the outer rarity_* span
        // (the item's name) and not the same-classed rarity_* span
        // nested inside item_tooltip (that one's a quality percentage,
        // already covered separately by the item_tooltip call above).
        applyItemNames('.equipment_slot_div > [class*="rarity_"]');
        // Bestiary/levelary loot info — both reuse the same
        // bestiary_entry_tooltip class (one shared list component for
        // both), but format loot differently: bestiary wraps each name
        // in its own .loot_name div (plain exact-match), while
        // levelary is raw "[ Name ] - X%" text lines with no wrapper,
        // caught by the bracket pattern instead. Scoping to the whole
        // tooltip (not just .loot_name) picks up both at once. Not
        // container-scoped like the inventory calls above — this
        // tooltip may detach/reattach elsewhere in the DOM on hover
        // rather than staying nested under #bestiary_list/#levelary_list,
        // so scoping to a container here risks silently missing it.
        // Safe either way since matching is content-based, not a
        // persisted "already done" flag.
        applyItemNames('.bestiary_entry_tooltip');
        // Event/message log — quote-wrapped item names within entries
        // like loot drops. Scoped to the whole log container (not just
        // .message_loot) in case other message types reference items
        // too. Unlike everything else scoped so far, this container
        // can grow unboundedly over a long play session rather than
        // staying at a fixed size — should be fine at any reasonable
        // log length, but if a slowdown ever shows up specifically
        // tied to very long sessions with a huge accumulated log,
        // this is the first place to look.
        applyItemNames('#message_box_div');
        // Gathering tooltip's drop-chance items ("百年柳木" etc) are
        // quote-wrapped — same quote pattern as the message log, no
        // new matching logic needed, just this call. The gathering
        // skill name is also quote-wrapped in the same tooltip
        // (e.g. "砍伐"), but isn't in itemNames, so the pattern safely
        // no-ops on it there; the buttons-mechanism entry above
        // handles that one instead.
        applyItemNames('#gathering_tooltip');
    }

    scan();
    document.addEventListener('DOMContentLoaded', scan);

    // Re-scan when new nodes appear (menus/dialogs opening, tabs
    // switching, or content getting rebuilt), batched to at most once
    // per animation frame.
    let queued = false;
    function scheduleScan() {
        if (queued) return;
        queued = true;
        requestAnimationFrame(() => { queued = false; scan(); });
    }

    new MutationObserver(scheduleScan).observe(document.documentElement, {
        childList: true,
        subtree: true,
    });
})();
